set(DSACPP_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(DSACPP_COMMON_DIR ${DSACPP_SOURCE_DIR}/common)
set(DSACPP_CHAPTERS ch01 ch02 ch03 ch04 ch05 ch06 ch07 ch08 ch09 ch10 ch11 ch12)
set(DSACPP_CHAPTER_INCLUDE_DIRS ${DSACPP_COMMON_DIR} ${DSACPP_SOURCE_DIR})
foreach(chapter IN LISTS DSACPP_CHAPTERS)
    list(APPEND DSACPP_CHAPTER_INCLUDE_DIRS ${DSACPP_SOURCE_DIR}/${chapter})
endforeach()

function(dsacpp_configure_target target)
    target_include_directories(${target} PRIVATE ${DSACPP_CHAPTER_INCLUDE_DIRS})
endfunction()

foreach(chapter IN LISTS DSACPP_CHAPTERS)
    add_subdirectory(${chapter})
endforeach()
